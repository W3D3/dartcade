# X01 Live Stats Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The X01 match screen's averages, last visit, Chalkboard and a new checkout rate come
from the server's game state, so they survive a reload or reconnect.

**Architecture:** The X01 reducer (`backend/src/games/x01.ts`) records every committed visit and
counts checkout darts; its view sends them with each snapshot (`schema/game-ws-v1.json`, then
`npm run gen:api`). The frontend (`playerStats.ts`, `teams.ts`, `GameDisplay.svelte`) reads the
snapshot instead of rebuilding history in the browser. Around the Clock keeps `visitHistory.ts`.

**Tech Stack:** TypeScript, Vitest, Svelte 5, JSON Schema → generated TS types and zod
(`scripts/gen-api.mjs`).

**Spec:** `docs/superpowers/specs/2026-10-07-x01-live-stats-design.md`

## Global Constraints

- Checkout attempt: a dart thrown by an opened player whose score before it is a one-dart finish
  (`checkoutHint(score, outMode, 1) !== null`), wherever it lands. Hit: that dart finished the leg.
- Checkout counts are per seat, over the whole match; never reset per leg.
- `legVisits` resets every leg; `lastVisit` does not.
- The match average divides by darts in committed visits (`dartsThrown`), never by `totalDarts`
  (it includes the open visit).
- Schema changes are additive only; run `npm run gen:api` from the repo root and commit its output.
- Commit messages: `type(scope): subject`, lower case, imperative, under 72 characters, ending with
  the session's `Co-Authored-By` / `Claude-Session` trailer lines.
- Formatting: `npm run format` at the repo root before committing.

## Review Focus

1. A reload mid-visit: the match average must not count the open visit's darts — Task 1 tests
   `dartsThrown` stays put until the takeout.
2. Undo of a committed visit (nothing open → `reopen`): the last visit, Chalkboard and counters
   go back with it — Task 3 adds an `applyInput` test.
3. Double in, not yet opened: those darts count as thrown (they were), score nothing, and are
   never checkout attempts — Tasks 1 and 2.
4. The leg-winning visit: `lastVisit` keeps it into the next leg, `legVisits` starts empty, and
   it adds only the darts actually thrown (1 or 2 for an early checkout) — Task 1.
5. Teams: the score is per team but last visit, darts and checkout counts are per seat, and the
   team Chalkboard keeps throw order across the team's seats — Tasks 2 and 4.

// Ends the open visit for its thrower: its record and the per-player totals that count it
function committedVisit(s: X01State): Pick<X01State, 'pointsScored' | 'dartsThrown' | 'legVisits' | 'lastVisit' | 'visitDarts'> {
  const cp = s.currentPlayer
  const t = s.teamOf[cp]
  const record: X01VisitRecord = {
    seat: cp,
    scored: s.visitOpenedScores[t] - s.scores[t],
    left: s.scores[t],
    bust: s.bustThisVisit,
    darts: s.visitDarts,
  }
  return {
    pointsScored: withVisitScored(s),
    dartsThrown: s.dartsThrown.map((d, i) => (i === cp ? d + s.visitDarts : d)),
    legVisits: [...s.legVisits, record],
    lastVisit: s.lastVisit.map((v, i) => (i === cp ? record : v)),
    visitDarts: 0,
  }
}

        const dart = e.data.dart
        // The thrower's team: its score counts down
        const t = s.teamOf[s.currentPlayer]

        if (s.bustThisVisit) return { state: s }
        if (s.scores[t] === 0) return { state: s }

        const thrown = { ...s, visitDarts: s.visitDarts + 1 }

        if (!s.opened[t]) {
          const opens = meetsMode(dart, s.cfg.inMode)
          if (!opens) return { state: thrown }
          const opened = s.opened.map((o, i) => (i === t ? true : o))
          const dartScore = effectiveDartScore(dart, s.cfg.bullValue)
          const newScore = s.scores[t] - dartScore
          if (newScore < 0 || deadEnd(newScore, s.cfg.outMode) || (newScore === 0 && !meetsMode(dart, s.cfg.outMode))) {
            return {
              state: { ...thrown, opened, bustThisVisit: true, scores: s.scores.map((sc, i) => (i === t ? s.visitOpenedScores[t] : sc)) },
            }
          }
          return { state: { ...thrown, opened, scores: s.scores.map((sc, i) => (i === t ? newScore : sc)) } }
        }

        const dartScore = effectiveDartScore(dart, s.cfg.bullValue)
        const newScore = s.scores[t] - dartScore

        if (newScore < 0 || deadEnd(newScore, s.cfg.outMode) || (newScore === 0 && !meetsMode(dart, s.cfg.outMode))) {
          return { state: { ...thrown, scores: s.scores.map((sc, i) => (i === t ? s.visitOpenedScores[t] : sc)), bustThisVisit: true } }

        const cp = s.currentPlayer
        const t = s.teamOf[cp]
        const done = committedVisit(s)

        // Leg win, for the thrower's team; the checkout counts for the thrower
        if (s.scores[t] === 0) {
          const legs = s.legs.map((l, i) => (i === t ? l + 1 : l))
          const bestCheckout = s.bestCheckout.map((b, i) => (i === cp ? Math.max(b, s.visitOpenedScores[t]) : b))

        if (round > s.cfg.maxRounds) {
          return {

export function x01Player(
  game: X01Game,
  i: number,
  o: { active: boolean; suggest: boolean; checkout?: CheckoutPrefs; bust?: boolean; scoreUpdates?: ScoreUpdates },
): X01PlayerView {
  const remaining = game.scores.at(i) ?? 0
  const opened = game.opened.at(i) ?? true
  const running = o.active ? game.currentVisitDarts : []
  const dartsLeft = 3 - running.length
  const hint =
    o.suggest && !o.bust && opened && remaining > 0 && dartsLeft > 0
      ? checkoutHint(remaining, game.config.outMode, dartsLeft, o.checkout)
      : null
  const leg: Visit[] = game.legVisits.filter(v => v.seat === i)
  const last = game.lastVisit.at(i) ?? null
  return {
    remaining,
    opened,
    shown: shownScore({
      mode: o.scoreUpdates ?? 'dart',
      score: remaining,
      thrower: o.active,
      locked: game.visitLocked,
      darts: running,
      start: game.visitStartScores.at(i) ?? null,
    }),
    canFinish: hint ? hint.join(' · ') : null,
    avg: avgOf(game.pointsScored.at(i) ?? 0, game.dartsThrown.at(i) ?? 0),
    legAvg: avgOf(
      leg.reduce((a, v) => a + v.scored, 0),
      leg.reduce((a, v) => a + v.darts, 0),
    ),
    last: last ? String(last.scored) : '—',
    ...checkoutRate(game.checkoutHits.at(i) ?? 0, game.checkoutAttempts.at(i) ?? 0),
    darts: game.totalDarts.at(i) ?? 0,
    legsWon: game.legs.at(i) ?? 0,
    firstTo: game.firstTo,
    leg: game.legs.reduce((a, l) => a + l, 0),
    visits: leg,
    showFinish: o.suggest,
    current: running.length ? { scored: running.reduce((a, d) => a + d.score, 0), left: remaining, bust: o.bust === true } : null,
  }
}
